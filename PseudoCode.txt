ALGORITHM Sampark_Saathi_Emergency_Pipeline:

    FUNCTION Main():
        // 1. INPUT: Capture Spoken Emergency & User Location
        user_audio_file = RECORD_AUDIO_FROM_MIC()
        user_lat, user_long = GET_USER_GEOLOCATION()
        
        IF user_audio_file IS EMPTY THEN
            DISPLAY "Error: Please record or state your emergency."
            RETURN
        END IF

        // 2. SPEECH-TO-TEXT: Transcribe Local Regional Audio
        transcribed_text = CONVERT_SPEECH_TO_TEXT(user_audio_file, target_lang="English")

        // 3. MULTI-EMERGENCY AI TRIAGE: Pass Text + Location to Gemma 4
        triage_result = CALL_GEMMA_4_MODEL(transcribed_text, user_lat, user_long)

        // Extract JSON fields from Gemma 4
        category        = triage_result["category"]        // "MEDICAL" | "FIRE" | "POLICE_CRIME" | "NATURAL_DISASTER"
        urgency_level   = triage_result["urgency"]         // "CRITICAL" | "HIGH" | "MEDIUM"
        english_summary = triage_result["summary_en"]
        local_advice    = triage_result["advice_local"]
        facility_needed = triage_result["facility_type"]   // "Hospital" | "Fire Station" | "Police Post"

        // 4. DYNAMIC FACILITY ROUTING: Find Nearest Support Location
        nearest_station = FIND_NEAREST_FACILITY(user_lat, user_long, facility_needed)

        // 5. TEXT-TO-SPEECH: Generate Local Dialect Audio Guidance
        response_audio_file = CONVERT_TEXT_TO_SPEECH(local_advice, lang="Regional_Language")

        // 6. OUTPUT: Update Dashboard & Auto-Play Guidance
        DISPLAY_EMERGENCY_CARD(category, urgency_level, english_summary, nearest_station)
        PLAY_AUDIO_RESPONSE(response_audio_file)

    END FUNCTION


    // Helper Function: Multi-Emergency Gemma 4 API Request
    FUNCTION CALL_GEMMA_4_MODEL(text_input, lat, long):
        system_prompt = """
        You are 'Sampark Saathi', a multilingual emergency assistant for rural India.
        Analyze the incoming voice transcription and categorize the incident.
        
        Return ONLY raw JSON with:
        - category: "MEDICAL" | "FIRE" | "POLICE_CRIME" | "NATURAL_DISASTER"
        - urgency: "CRITICAL" | "HIGH" | "MEDIUM"
        - summary_en: 1-sentence English description of the incident
        - advice_local: Clear, immediate action instructions translated into user's local dialect
        - facility_type: Recommended assistance hub (e.g. "Hospital", "Fire Station", "Police Post")
        """

        payload = FORMAT_PROMPT(system_prompt, text_input, lat, long)
        json_response = GEMMA_4_API_REQUEST(payload)
        
        RETURN PARSE_JSON(json_response)
    END FUNCTION
